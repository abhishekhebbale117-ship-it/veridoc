# ============================================================ # STUDENT RESULT / SUBMISSION STATUS # ============================================================ @app.route("/result") def result(): # -------------------------------------------------------- # LOGIN REQUIRED # -------------------------------------------------------- if "user_id" not in session: flash( "Please login first." ) return redirect( url_for("login") ) # -------------------------------------------------------- # ADMIN -> ADMIN DASHBOARD # -------------------------------------------------------- if session.get("user_role") == "admin": return redirect( url_for("admin_dashboard") ) db = get_db() # -------------------------------------------------------- # GET LAST SUBMITTED BATCH # -------------------------------------------------------- batch_id = session.get( "last_batch_id" ) # -------------------------------------------------------- # IF SESSION DOES NOT HAVE BATCH ID, # FIND THE LATEST BATCH FROM DATABASE # -------------------------------------------------------- if not batch_id: latest = db.execute(""" SELECT batch_id FROM verification_requests WHERE user_id = ? AND batch_id IS NOT NULL AND batch_id != '' ORDER BY id DESC LIMIT 1 """, ( session["user_id"], )).fetchone() if latest: batch_id = latest["batch_id"] session[ "last_batch_id" ] = batch_id # -------------------------------------------------------- # NO SUBMISSION # -------------------------------------------------------- if not batch_id: db.close() return render_template( "result.html", submissions=[], approval_status="NONE", approved_count=0, pending_count=0, rejected_count=0, batch_id=None ) # -------------------------------------------------------- # GET ALL DOCUMENTS FROM THIS BATCH # -------------------------------------------------------- submissions = db.execute(""" SELECT * FROM verification_requests WHERE user_id = ? AND batch_id = ? ORDER BY id ASC """, ( session["user_id"], batch_id )).fetchall() db.close() # -------------------------------------------------------- # DETERMINE OVERALL STATUS # -------------------------------------------------------- approval_status = "PENDING" if submissions: statuses = [ ( row["approval_status"] or "PENDING" ).upper() for row in submissions ] # ALL APPROVED if all( status == "APPROVED" for status in statuses ): approval_status = "APPROVED" # ANY REJECTED elif any( status == "REJECTED" for status in statuses ): approval_status = "REJECTED" # OTHERWISE PENDING else: approval_status = "PENDING" # -------------------------------------------------------- # COUNTS # -------------------------------------------------------- approved_count = sum( 1 for row in submissions if ( row["approval_status"] or "PENDING" ).upper() == "APPROVED" ) pending_count = sum( 1 for row in submissions if ( row["approval_status"] or "PENDING" ).upper() == "PENDING" ) rejected_count = sum( 1 for row in submissions if ( row["approval_status"] or "PENDING" ).upper() == "REJECTED" ) # -------------------------------------------------------- # SEND CORRECT VARIABLES TO RESULT.HTML # -------------------------------------------------------- return render_template( "result.html", submissions=submissions, approval_status=approval_status, approved_count=approved_count, pending_count=pending_count, rejected_count=rejected_count, batch_id=batch_id )
